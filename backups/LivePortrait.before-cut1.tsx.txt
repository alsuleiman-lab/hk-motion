const BeatOffset = React.createContext(0);
const useAbs = () => useCurrentFrame() + React.useContext(BeatOffset);
const usePop = (at: number, damping = 12) => spring({ frame: useAbs() - sec(at), fps, config: { damping, stiffness: 170, mass: 0.7 } });
const Pop: React.FC<{ at: number; children: React.ReactNode; style?: React.CSSProperties; from?: number }> = ({ at, children, style, from = 0.6 }) => {
  const p = usePop(at);
  return <div style={{ opacity: Math.min(1, p * 2.2), transform: `scale(${from + (1 - from) * p})`, ...style }}>{children}</div>;
};
const At: React.FC<{ x: number; y: number; w: number; children: React.ReactNode }> = ({ x, y, w, children }) => (
  <div style={{ position: "absolute", left: x, top: y, width: w, display: "flex", justifyContent: "center" }}>{children}</div>
);

const Dots: React.FC<{ color: string }> = ({ color }) => {

};

const Photo: React.FC<{ src: string; w: number; h: number; at: number; rot?: number; pos?: string }> = ({ src, w, h, at, rot = 0, pos = "50% 50%" }) => {
  const f = useAbs();

      </div>
    </Pop>
  );
};

// ---- speaker -------------------------------------------------------------------------------
type Rect = { x: number; y: number; w: number; h: number; r: number };
const rectFor = (l: Layout): Rect => (l === "fs" ? { x: 0, y: 0, w: W, h: H, r: 0 } : { x: 130, y: 110, w: 820, h: 640, r: 48 });
const lerp = (a: number, b: number, p: number) => a + (b - a) * p;
const PHASE_B = 24.1; // after the embedded clip panel disappears the speaker is centred in the source frame

const Speaker: React.FC = () => {
  const f = useCurrentFrame();
  const t = f / fps + SHIFT;
  let i = beats.findIndex((b) => t >= b.from && t < b.to);
  if (i < 0) i = beats.length - 1;
  const cur = rectFor(beats[i].layout);
  const prev = i > 0 ? rectFor(beats[i - 1].layout) : cur;
  const p = Easing.out(Easing.cubic)(interpolate(f - sec(beats[i].from - SHIFT), [0, 12], [0, 1], { extrapolateLeft: "clamp", extrapolateRight: "clamp" }));
  const r = { x: lerp(prev.x, cur.x, p), y: lerp(prev.y, cur.y, p), w: lerp(prev.w, cur.w, p), h: lerp(prev.h, cur.h, p), r: lerp(prev.r, cur.r, p) };
  const q = Math.min(1, Math.max(0, (r.w - 820) / (W - 820)));
  const s = lerp(0.82, 1.78, q);
  const cx = t < PHASE_B ? 1485 : 915;
  const cy = t < PHASE_B ? 405 : 300;
  const ty = r.h * lerp(0.54, 0.28, q);
  const left = Math.min(0, Math.max(r.w - 1920 * s, r.w / 2 - cx * s));
  const top = Math.min(0, Math.max(r.h - 1080 * s, ty - cy * s));
  return (
    <div style={{ position: "absolute", left: r.x, top: r.y, width: r.w, height: r.h, borderRadius: r.r, overflow: "hidden", boxShadow: r.r > 0 ? "0 16px 44px rgba(0,0,0,.5)" : undefined, border: r.r > 0 ? "6px solid #fff" : undefined, background: "#000" }}>
      <OffthreadVideo src={staticFile("live_cut_v12.mp4")} style={{ position: "absolute", left, top, width: 1920 * s, height: 1080 * s, maxWidth: "none" }} />
      <AbsoluteFill style={{ background: "linear-gradient(180deg, rgba(0,0,0,0) 62%, rgba(0,0,0,.4) 100%)", opacity: q }} />
    </div>
  );
};
